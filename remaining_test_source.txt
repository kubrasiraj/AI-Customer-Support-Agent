from datetime import datetime, timedelta, timezone

from jose import JWTError, jwt
from passlib.context import CryptContext

from app.core.config import settings


pwd_context = CryptContext(
    schemes=["bcrypt"],
    deprecated="auto",
)


def hash_password(password: str) -> str:
    """Convert a plain password into a secure hash."""
    return pwd_context.hash(password)


def verify_password(
    plain_password: str,
    hashed_password: str,
) -> bool:
    """Check whether a plain password matches its stored hash."""
    return pwd_context.verify(
        plain_password,
        hashed_password,
    )


def create_access_token(data: dict) -> str:
    """Create a JWT access token."""
    to_encode = data.copy()

    expire = datetime.now(timezone.utc) + timedelta(
        minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES
    )

    to_encode.update({"exp": expire})

    return jwt.encode(
        to_encode,
        settings.SECRET_KEY,
        algorithm=settings.ALGORITHM,
    )


def decode_access_token(token: str) -> dict:
    """Decode and validate a JWT access token."""
    try:
        payload = jwt.decode(
            token,
            settings.SECRET_KEY,
            algorithms=[settings.ALGORITHM],
        )

        return payload

    except JWTError:
        raise ValueError("Invalid or expired token")

def compress_context(
    results: list[dict],
    max_chars: int = 6000,
) -> str:
    """
    Combine retrieved chunks into a limited-size context.

    This prevents unnecessarily large context from
    being sent to the LLM.
    """

    if not results:
        return ""

    context_parts = []
    current_length = 0

    for result in results:
        content = result["content"].strip()

        if not content:
            continue

        chunk_text = (
            f"[Knowledge Chunk {result['chunk_id']}]\n"
            f"{content}"
        )

        additional_length = len(chunk_text)

        if current_length + additional_length > max_chars:
            remaining = max_chars - current_length

            if remaining > 100:
                chunk_text = chunk_text[:remaining]
                context_parts.append(chunk_text)

            break

        context_parts.append(chunk_text)
        current_length += additional_length

    return "\n\n".join(context_parts)

from sqlalchemy.orm import Session

from app.services.semantic_search import semantic_search
from app.rag.keyword_search import keyword_search


def hybrid_search(
    db: Session,
    query: str,
    top_k: int = 5,
) -> list[dict]:
    """
    Combine semantic search and keyword search.

    Semantic search weight: 70%
    Keyword search weight: 30%
    """

    if not query.strip():
        raise ValueError("Query cannot be empty")

    semantic_results = semantic_search(
        db=db,
        query=query,
        top_k=top_k,
    )

    keyword_results = keyword_search(
        db=db,
        query=query,
        top_k=top_k,
    )

    combined = {}

    # Add semantic search results
    for result in semantic_results:
        chunk_id = result["chunk_id"]

        combined[chunk_id] = {
            "chunk_id": chunk_id,
            "document_id": result["document_id"],
            "chunk_index": result["chunk_index"],
            "content": result["content"],
            "semantic_score": result["score"],
            "keyword_score": 0.0,
        }

    # Add keyword search results
    for result in keyword_results:
        chunk_id = result["chunk_id"]

        if chunk_id not in combined:
            combined[chunk_id] = {
                "chunk_id": chunk_id,
                "document_id": result["document_id"],
                "chunk_index": result["chunk_index"],
                "content": result["content"],
                "semantic_score": 0.0,
                "keyword_score": result["keyword_score"],
            }
        else:
            combined[chunk_id]["keyword_score"] = result["keyword_score"]

    # Calculate hybrid score
    results = []

    for result in combined.values():
        hybrid_score = (
            0.70 * result["semantic_score"]
            + 0.30 * result["keyword_score"]
        )

        result["hybrid_score"] = hybrid_score

        results.append(result)

    # Highest hybrid score first
    results.sort(
        key=lambda item: item["hybrid_score"],
        reverse=True,
    )

    return results[:top_k]

from sqlalchemy.orm import Session

from app.models.knowledge_chunk import KnowledgeChunk


def keyword_search(
    db: Session,
    query: str,
    top_k: int = 5,
) -> list[dict]:
    """
    Find knowledge chunks using simple keyword matching.
    """

    if not query.strip():
        raise ValueError("Query cannot be empty")

    query_words = set(query.lower().split())

    chunks = (
        db.query(KnowledgeChunk)
        .filter(KnowledgeChunk.embedding.isnot(None))
        .all()
    )

    results = []

    for chunk in chunks:
        content_words = set(chunk.content.lower().split())

        matched_words = query_words.intersection(content_words)

        if not matched_words:
            continue

        score = len(matched_words) / len(query_words)

        results.append(
            {
                "chunk_id": chunk.id,
                "document_id": chunk.document_id,
                "chunk_index": chunk.chunk_index,
                "content": chunk.content,
                "keyword_score": score,
            }
        )

    results.sort(
        key=lambda item: item["keyword_score"],
        reverse=True,
    )

    return results[:top_k]

import re


def transform_query(question: str) -> str:
    """
    Convert a customer question into a simple
    retrieval-friendly query.

    Temporary deterministic version.
    Later we can replace this with an LLM-based
    query transformer.
    """

    if not question.strip():
        raise ValueError("Question cannot be empty")

    query = question.lower().strip()

    # Remove common conversational phrases
    phrases_to_remove = [
        "can i",
        "could i",
        "would i",
        "please",
        "i want to know",
        "i want to",
        "what can i do",
        "what should i do",
        "tell me",
        "is it possible",
    ]

    for phrase in phrases_to_remove:
        query = query.replace(phrase, "")

    # Remove unnecessary punctuation
    query = re.sub(r"[^\w\s]", " ", query)

    # Normalize whitespace
    query = re.sub(r"\s+", " ", query).strip()

    return query


========== app\rag\reranker.py ==========
from sentence_transformers import CrossEncoder


MODEL_NAME = "cross-encoder/ms-marco-MiniLM-L-6-v2"

_reranker = None


def get_reranker() -> CrossEncoder:
    """
    Load the reranker model only when it is needed.
    """

    global _reranker

    if _reranker is None:
        _reranker = CrossEncoder(MODEL_NAME)

    return _reranker


def rerank_results(
    query: str,
    results: list[dict],
    top_k: int = 3,
) -> list[dict]:
    """
    Re-rank retrieved documents using a cross-encoder.
    """

    if not query.strip():
        raise ValueError("Query cannot be empty")

    if not results:
        return []

    reranker = get_reranker()

    pairs = [
        (query, result["content"])
        for result in results
    ]

    scores = reranker.predict(pairs)

    reranked_results = []

    for result, score in zip(results, scores):
        updated_result = result.copy()

        updated_result["rerank_score"] = float(score)

        reranked_results.append(updated_result)

    reranked_results.sort(
        key=lambda item: item["rerank_score"],
        reverse=True,
    )

    return reranked_results[:top_k]

from fastapi import (

from app.models.user import User
from app.schemas.knowledge import KnowledgeDocumentResponse
from app.services.knowledge_service import process_knowledge_document


router = APIRouter(
    prefix="/knowledge",
    tags=["Knowledge Base"],
)


UPLOAD_DIR = Path("app/knowledge")

ALLOWED_EXTENSIONS = {
    ".txt",
}


@router.post(
    "/documents",
    response_model=KnowledgeDocumentResponse,
    status_code=status.HTTP_201_CREATED,
)
def upload_knowledge_document(
    background_tasks: BackgroundTasks,
    file: UploadFile = File(...),
    title: str = Form(...),
    category: str = Form(...),
    description: str | None = Form(None),
    version: str = Form("1.0"),
    current_user: User = Depends(
        require_role("admin", "support")
    ),
    db: Session = Depends(get_db),
):
    """
    Upload a company knowledge document.

    Only admin and support users can add
    documents to the knowledge base.
    """

    # Check file extension.
    extension = Path(
        file.filename or ""
    ).suffix.lower()

    if extension not in ALLOWED_EXTENSIONS:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Only .txt files are currently supported",
        )

    # Make sure the knowledge directory exists.
    UPLOAD_DIR.mkdir(
        parents=True,
        exist_ok=True,
    )

    # Generate a safe file name.
    file_name = Path(
        file.filename or "document.txt"
    ).name

    file_path = UPLOAD_DIR / file_name

    # Save uploaded file.
    with file_path.open("wb") as buffer:
        buffer.write(
            file.file.read()
        )

    # Create database document record.
    document = KnowledgeDocument(
        title=title,
        file_name=file_name,
        file_path=str(file_path),
        category=category,
        description=description,
        version=version,
        status="uploaded",
    )

    db.add(document)
    db.commit()
    db.refresh(document)

    # Process the document in the background.
    background_tasks.add_task(
        process_knowledge_document,
        db,
        document,
    )

    return document

def get_customer_for_user(
    db: Session,
    user_id: int,
) -> Customer | None:
    """Find the customer profile linked to the authenticated user."""

    return (
        db.query(Customer)
        .filter(Customer.user_id == user_id)
        .first()
    )


@router.post(
    "",
    response_model=OrderResponse,
    status_code=status.HTTP_201_CREATED,
)
def create_order(
    data: OrderCreate,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """Create an order for the authenticated customer."""

    customer = get_customer_for_user(
        db,
        current_user.id,
    )

    if not customer:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Customer profile not found",
        )

    try:
        order = create_order_service(
            db,
            current_user.id,
            data,
        )

        return order

    except ValueError as exc:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=str(exc),
        )


@router.get(
    "",
    response_model=list[OrderResponse],
)
def get_my_orders(
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """Return only the orders belonging to the authenticated customer."""

    customer = get_customer_for_user(
        db,
        current_user.id,
    )

    if not customer:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Customer profile not found",
        )

    orders = (
        db.query(Order)
        .options(selectinload(Order.items))
        .filter(Order.customer_id == customer.id)
        .order_by(Order.created_at.desc())
        .all()
    )

    return orders


@router.get(
    "/{order_id}",
    response_model=OrderResponse,
)
def get_order(
    order_id: int,
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
):
    """Return one order only if it belongs to the authenticated customer."""

    customer = get_customer_for_user(
        db,
        current_user.id,
    )

    if not customer:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Customer profile not found",
        )

    order = (
        db.query(Order)
        .options(selectinload(Order.items))
        .filter(
            Order.id == order_id,
            Order.customer_id == customer.id,
        )
        .first()
    )

    if not order:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Order not found",
        )

    return order

def run_support_agent(
    db: Session,
    user_id: int,
    question: str,
    conversation_id: int | None = None,
) -> dict:
    if not question.strip():
        raise HTTPException(
            status_code=422,
            detail="Question cannot be empty.",
        )

    # 1. Resolve the authenticated user to their customer record.
    customer = (
        db.query(Customer)
        .filter(Customer.user_id == user_id)
        .first()
    )

    if customer is None:
        raise HTTPException(
            status_code=404,
            detail="No customer profile is linked to this account.",
        )

    # 2. Load the requested conversation or create a new one.
    if conversation_id is None:
        conversation = Conversation(
            customer_id=customer.id,
            status="active",
        )
        db.add(conversation)
        db.flush()

    else:
        conversation = (
            db.query(Conversation)
            .filter(
                Conversation.id == conversation_id,
                Conversation.customer_id == customer.id,
            )
            .first()
        )

        if conversation is None:
            # Do not reveal whether another customer's conversation exists.
            raise HTTPException(
                status_code=404,
                detail="Conversation not found.",
            )

        if conversation.status != "active":
            raise HTTPException(
                status_code=409,
                detail="This conversation is not active.",
            )

    # 3. Load previous messages in chronological order.
    previous_messages = (
        db.query(Message)
        .filter(Message.conversation_id == conversation.id)
        .order_by(Message.created_at.asc(), Message.id.asc())
        .all()
    )

    # 4. Save the new user message.
    user_message = Message(
        conversation_id=conversation.id,
        role="user",
        content=question.strip(),
    )
    db.add(user_message)
    db.flush()

    # 5. Convert stored messages to LangChain's message format.
    history = []

    for message in previous_messages:
        if message.role == "user":
            history.append({
                "role": "user",
                "content": message.content,
            })
        elif message.role in ("assistant", "ai"):
            history.append({
                "role": "assistant",
                "content": message.content,
            })

    # Add the current question after the earlier history.
    history.append({
        "role": "user",
        "content": question.strip(),
    })

    try:
        tools = create_database_tools(
            db=db,
            authenticated_customer_id=customer.id,

        )
        agent = create_support_agent(tools)

        # 6. Invoke the agent with the complete conversation history.
        result = agent.invoke({
            "messages": history,
        })

        messages = result.get("messages", [])

        if not messages:
            raise RuntimeError(
                "The agent returned no messages."
            )

        final_message = messages[-1]
        content = final_message.content

        if isinstance(content, str):
            answer = content
        else:
            answer = "".join(
                item.get("text", "")
                for item in content
                if isinstance(item, dict)
                and item.get("type") == "text"
            ).strip()

        if not answer:
            raise RuntimeError(
                "The agent returned an empty answer."
            )

        # 7. Collect tool names for API transparency and testing.
        tools_used = []

        for message in messages:
            for tool_call in getattr(message, "tool_calls", []):
                tool_name = tool_call.get("name")

                if tool_name and tool_name not in tools_used:
                    tools_used.append(tool_name)

        # 8. Save the assistant's final response.
        assistant_message = Message(
            conversation_id=conversation.id,
            role="assistant",
            content=answer,
        )
        db.add(assistant_message)

        conversation.updated_at = __import__(
            "datetime"
        ).datetime.now(
            __import__("datetime").timezone.utc
        )

        # Persist the conversation and both messages.
        db.commit()

        return {
            "answer": answer,
            "conversation_id": conversation.id,
            "tools_used": tools_used,
        }

    except Exception:
        db.rollback()
        raise

from pathlib import Path


def extract_text(file_path: str) -> str:
    """
    Read a text file and return its content.
    """

    path = Path(file_path)

    if not path.exists():
        raise FileNotFoundError(
            f"File not found: {file_path}"
        )

    text = path.read_text(
        encoding="utf-8"
    )

    if not text.strip():
        raise ValueError(
            "Document is empty"
        )

    return text

from sqlalchemy.orm import Session

from app.models.knowledge_document import KnowledgeDocument
from app.models.knowledge_chunk import KnowledgeChunk
from app.services.document_parser import extract_text


def split_text_into_chunks(
    text: str,
    chunk_size: int = 500,
    chunk_overlap: int = 50,
) -> list[str]:
    """
    Split text into smaller overlapping chunks.
    """

    if not text.strip():
        return []

    if chunk_overlap >= chunk_size:
        raise ValueError(
            "chunk_overlap must be smaller than chunk_size"
        )

    words = text.split()

    chunks = []

    start = 0

    while start < len(words):
        end = start + chunk_size

        chunk_words = words[start:end]

        chunk = " ".join(chunk_words).strip()

        if chunk:
            chunks.append(chunk)

        start += chunk_size - chunk_overlap

    return chunks


def process_knowledge_document(
    db: Session,
    document: KnowledgeDocument,
) -> int:
    """
    Extract text from a knowledge document,
    split it into chunks, and save the chunks
    in the database.
    """

    # Mark document as processing.
    document.status = "processing"
    db.commit()

    try:
        # Extract the original document text.
        text = extract_text(document.file_path)

        # Split document into smaller chunks.
        chunks = split_text_into_chunks(text)

        if not chunks:
            raise ValueError(
                "No text could be extracted from the document"
            )

        # Remove old chunks if the document is being reprocessed.
        db.query(KnowledgeChunk).filter(
            KnowledgeChunk.document_id == document.id
        ).delete(
            synchronize_session=False
        )

        # Save each chunk.
        for index, content in enumerate(chunks):
            chunk = KnowledgeChunk(
                document_id=document.id,
                chunk_index=index,
                content=content,
            )

            db.add(chunk)

        # Mark document as successfully processed.
        document.status = "processed"

        db.commit()

        return len(chunks)

    except Exception:
        # Mark document as failed if anything goes wrong.
        db.rollback()

        document.status = "failed"
        db.commit()

        raise

def answer_customer_question(
    db: Session,
    question: str,
) -> dict:
    """
    Complete Advanced RAG pipeline.

    Flow:

    Customer Question
        ↓
    Query Transformation
        ↓
    Hybrid Search
        ↓
    Reranking
        ↓
    Context Compression
        ↓
    LangChain + Gemini
        ↓
    Final Answer
    """

    if not question.strip():
        raise ValueError("Question cannot be empty")

    # -------------------------------------------------
    # 1. Transform the customer question
    # -------------------------------------------------

    retrieval_query = transform_query(question)

    # -------------------------------------------------
    # 2. Hybrid retrieval
    # -------------------------------------------------

    retrieved_results = hybrid_search(
        db=db,
        query=retrieval_query,
        top_k=5,
    )

    if not retrieved_results:
        return {
            "answer": (
                "I could not find enough information "
                "in the support knowledge base to answer "
                "your question."
            ),
            "sources": [],
        }

    # -------------------------------------------------
    # 3. Rerank retrieved results
    # -------------------------------------------------

    reranked_results = rerank_results(
        query=question,
        results=retrieved_results,
        top_k=3,
    )

    # -------------------------------------------------
    # 4. Compress context
    # -------------------------------------------------

    context = compress_context(
        results=reranked_results,
        max_chars=6000,
    )

    if not context:
        return {
            "answer": (
                "I could not find enough information "
                "in the support knowledge base to answer "
                "your question."
            ),
            "sources": [],
        }

    # -------------------------------------------------
    # 5. Generate answer using LangChain
    # -------------------------------------------------

    chain = create_rag_chain()

    response = chain.invoke(
        {
            "context": context,
            "question": question,
        }
    )
    if isinstance(response.content, str):
        answer = response.content
    else:
        answer = "".join(
        item.get("text", "")
        for item in response.content
        if isinstance(item, dict) and item.get("type") == "text"
    ).strip()
    # -------------------------------------------------
    # 6. Prepare source information
    # -------------------------------------------------

    sources = []

    for result in reranked_results:
        sources.append(
            {
                "chunk_id": result["chunk_id"],
                "document_id": result["document_id"],
                "semantic_score": round(
                    result["semantic_score"],
                    4,
                ),
                "keyword_score": round(
                    result["keyword_score"],
                    4,
                ),
                "hybrid_score": round(
                    result["hybrid_score"],
                    4,
                ),
                "rerank_score": round(
                    result["rerank_score"],
                    4,
                ),
            }
        )

    return {
        "answer": answer,
        "sources": sources,
    }


def retrieve_customer_knowledge(
    db: Session,
    question: str,
) -> dict:
    """
    Retrieve relevant company policies and FAQs for the AI agent.

    This function retrieves evidence only. It does not call an LLM
    to generate a separate answer.
    """

    if not question.strip():
        return {
            "found": False,
            "context": "",
            "sources": [],
            "message": "The question cannot be empty.",
        }

    # 1. Transform the question for retrieval.
    retrieval_query = transform_query(question)

    # 2. Retrieve relevant chunks using hybrid search.
    retrieved_results = hybrid_search(
        db=db,
        query=retrieval_query,
        top_k=5,
    )

    if not retrieved_results:
        return {
            "found": False,
            "context": "",
            "sources": [],
            "message": (
                "No relevant information was found "
                "in the support knowledge base."
            ),
        }

    # 3. Rerank the retrieved chunks.
    reranked_results = rerank_results(
        query=question,
        results=retrieved_results,
        top_k=3,
    )

    if not reranked_results:
        return {
            "found": False,
            "context": "",
            "sources": [],
            "message": (
                "No sufficiently relevant knowledge chunks were found."
            ),
        }

    # 4. Compress the evidence to fit the agent's context.
    context = compress_context(
        results=reranked_results,
        max_chars=6000,
    )

    if not context or not context.strip():
        return {
            "found": False,
            "context": "",
            "sources": [],
            "message": (
                "Relevant policy evidence could not be extracted."
            ),
        }

    # 5. Return evidence and source identifiers to the agent.
    sources = [
        {
            "chunk_id": result["chunk_id"],
            "document_id": result["document_id"],
        }
        for result in reranked_results
    ]

    return {
        "found": True,
        "context": context,
        "sources": sources,
        "message": (
            "Retrieved evidence from the support knowledge base. "
            "Answer only claims supported by this evidence."
        ),
    }

import math

from sqlalchemy.orm import Session

from app.models.knowledge_chunk import KnowledgeChunk
from app.services.embedding_service import generate_embedding


def cosine_similarity(
    vector_a: list[float],
    vector_b: list[float],
) -> float:
    """
    Calculate cosine similarity between two vectors.
    """

    if len(vector_a) != len(vector_b):
        raise ValueError("Vectors must have the same dimensions")

    dot_product = sum(
        a * b
        for a, b in zip(vector_a, vector_b)
    )

    magnitude_a = math.sqrt(
        sum(a * a for a in vector_a)
    )

    magnitude_b = math.sqrt(
        sum(b * b for b in vector_b)
    )

    if magnitude_a == 0 or magnitude_b == 0:
        return 0.0

    return dot_product / (magnitude_a * magnitude_b)


def semantic_search(
    db: Session,
    query: str,
    top_k: int = 3,
) -> list[dict]:
    """
    Find the most relevant knowledge chunks
    for a user query.
    """

    if not query.strip():
        raise ValueError("Query cannot be empty")

    query_embedding = generate_embedding(query)

    chunks = (
        db.query(KnowledgeChunk)
        .filter(KnowledgeChunk.embedding.isnot(None))
        .all()
    )

    results = []

    for chunk in chunks:
        similarity = cosine_similarity(
            query_embedding,
            chunk.embedding,
        )

        results.append(
            {
                "chunk_id": chunk.id,
                "document_id": chunk.document_id,
                "chunk_index": chunk.chunk_index,
                "content": chunk.content,
                "score": similarity,
            }
        )

    results.sort(
        key=lambda item: item["score"],
        reverse=True,
    )

    return results[:top_k]

        "incorrect item",
        "incorrect product",
        "different product delivered",
    ]

    manufacturing_defect_phrases = [
        "manufacturing defect",
        "manufacturing defective",
        "factory defect",
        "product is defective",
        "defective product",
    ]

    if any(
        phrase in normalized_reason
        for phrase in damaged_phrases
    ):
        return "damaged"

    if any(
        phrase in normalized_reason
        for phrase in wrong_product_phrases
    ):
        return "wrong product"

    if any(
        phrase in normalized_reason
        for phrase in manufacturing_defect_phrases
    ):
        return "manufacturing defect"

    return None


def check_refund_eligibility(
    db: Session,
    order_number: str,
    reason: str,
    customer_id: int,
) -> dict:
    """
    Check refund eligibility using the available order data.

    This tool does not create, approve, or process refunds.
    """

    # Step 1: Find the order.
    order = (
        db.query(Order)
        .filter(
            Order.order_number == order_number.strip(),
            Order.customer_id == customer_id,
        )
        .first()
    )

    if order is None:
        return {
            "eligible": False,
            "reason": "Order not found.",
            "requires_human_approval": False,
        }

    # Step 2: Validate the refund reason.
    refund_category = normalize_refund_reason(reason)

    if refund_category is None:
        return {
            "eligible": False,
            "reason": (
                "The refund reason is not recognized. Supported "
                "reasons include damaged products, wrong products, "
                "and manufacturing defects."
            ),
            "requires_human_approval": False,
            "order_number": order.order_number,
        }

    # Step 3: Verify that the order is recorded as delivered.
    if order.status.strip().lower() != "delivered":
        return {
            "eligible": False,
            "reason": (
                "Refund eligibility cannot be confirmed because "
                "the order is not recorded as delivered. "
                f"Current order status: {order.status}."
            ),
            "requires_human_approval": False,
            "order_number": order.order_number,
            "refund_category": refund_category,
        }

    # Step 4: The current Order model has no delivery-date field.
    # Do not use created_at as a substitute for delivery date.
    return {
        "eligible": False,
        "reason": (
            "The order is marked as delivered, but the system does "
            "not store its delivery date. The 30-day refund window "
            "cannot be verified. Delivery-date verification is "
            "required before eligibility can be confirmed."
        ),
        "requires_human_approval": False,
        "order_number": order.order_number,
        "refund_category": refund_category,
    }

from sqlalchemy.orm import Session

from app.models.ticket import Ticket


def get_ticket(
    db: Session,
    ticket_id: int,
    customer_id: int,
) -> dict | None:
    """Return a ticket only if it belongs to the authenticated customer."""

    ticket = (
        db.query(Ticket)
        .filter(
            Ticket.id == ticket_id,
            Ticket.customer_id == customer_id,
        )
        .first()
    )

    if ticket is None:
        return None

    return {
        "ticket_id": ticket.id,
        "customer_id": ticket.customer_id,
        "status": ticket.status,
        "subject": ticket.subject,
        "description": ticket.description,
    }
